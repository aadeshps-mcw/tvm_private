

@T.prim_func
def tvmgen_default_fused_image_dilation2d(p0: T.Buffer((1, 64, 32, 32), "float32"), p1: T.Buffer((64, 3, 3), "float32"), compute: T.Buffer((1, 64, 32, 32), "float32")):
    T.func_attr({"from_legacy_te_schedule": T.bool(True), "tir.noalias": T.bool(True)})
    pad_temp = T.allocate([73984], "float32", "global")
    pad_temp_1 = T.Buffer((73984,), data=pad_temp)
    for i1, i2, i3 in T.grid(64, 34, 34):
        p0_1 = T.Buffer((65536,), data=p0.data)
        pad_temp_1[i1 * 1156 + i2 * 34 + i3] = T.if_then_else(1 <= i2 and i2 < 33 and 1 <= i3 and i3 < 33, p0_1[i1 * 1024 + i2 * 32 + i3 - 33], T.float32(0))
    for ff, yy, xx in T.grid(64, 32, 32):
        compute_1 = T.Buffer((65536,), data=compute.data)
        compute_1[ff * 1024 + yy * 32 + xx] = T.float32(-3.4028234663852886e+38)
        for ry, rx in T.grid(3, 3):
            p1_1 = T.Buffer((576,), data=p1.data)
            compute_1[ff * 1024 + yy * 32 + xx] = T.max(compute_1[ff * 1024 + yy * 32 + xx], pad_temp_1[ff * 1156 + yy * 34 + ry * 34 + xx + rx] + p1_1[ff * 9 + ry * 3 + rx])
